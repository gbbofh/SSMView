cmake_minimum_required(VERSION 3.18)
project(SSMView LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

find_package(raylib 5.0 QUIET)
if(NOT raylib_FOUND)
    include(FetchContent)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(raylib)
endif()

add_executable(ssmview src/main.c src/ssm.c)
target_include_directories(ssmview PRIVATE src)
target_link_libraries(ssmview PRIVATE raylib)
if(NOT MSVC)
    target_link_libraries(ssmview PRIVATE m)
    target_compile_options(ssmview PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(ssmparse tests/parse.c src/ssm.c)
    target_include_directories(ssmparse PRIVATE src)
    if(NOT MSVC)
        target_link_libraries(ssmparse PRIVATE m)
        target_compile_options(ssmparse PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
